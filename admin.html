(Остальной твой код, где открываются кейсы) function openCaseReward(type) { if (!userCases[type] || userCases[type].count <= 0) return; userCases[type].count--; saveCases(); renderCasesUI(); const modal = document.getElementById('case-modal'); if (!modal) return; modal.classList.add('show'); document.getElementById('modal-case-title').innerText = 'Открываем сундук...'; document.getElementById('modal-case-prize').innerText = '🎰'; document.getElementById('modal-case-text').innerText = 'Крутим барабан...'; document.getElementById('modal-case-btn').style.display = 'none'; setTimeout(() => { const drops = PRIZES[type] || []; if (drops.length === 0) { alert("В этом кейсе пока нет призов!"); return; } // Выбор приза по шансу const totalChance = drops.reduce((s, p) => s + p.chance, 0); let rand = Math.random() * totalChance; let prize = drops[0]; for (let p of drops) { if (rand < p.chance) { prize = p; break; } rand -= p.chance; } document.getElementById('modal-case-prize').innerText = prize.icon || '🎁'; document.getElementById('modal-case-title').innerText = 'Ваша награда!'; document.getElementById('modal-case-text').innerText = prize.name + "\n" + prize.desc; const btn = document.getElementById('modal-case-btn'); btn.style.display = 'block'; btn.onclick = () => { closeModal('case-modal'); showToast(`Вы получили: ${prize.name}`); }; }, 1500); }
